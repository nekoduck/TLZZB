phase('Leftovers')
const left = await parallel([
  () => agent(SEAMS, { label: 'seams', phase: 'Leftovers' }),
  () => agent(EVALS, { label: 'evals', phase: 'Leftovers' }),
])

phase('Regression review')
const reviews = (await parallel([
  () => agent(REVIEW_A, { label: 'regress:kernel-llm-game', phase: 'Regression review', schema: FINDINGS }),
  () => agent(REVIEW_B, { label: 'regress:play-world-content', phase: 'Regression review', schema: FINDINGS }),
])).filter(Boolean)

const all = reviews.flatMap(r => r.findings || [])
const todo = all.filter(f => f.severity !== 'low')
const low = all.filter(f => f.severity === 'low')
log('regression review: ' + all.length + ' findings, ' + todo.length + ' medium/high go to repair, ' + low.length + ' low only reported')

let repair = null
if (todo.length) {
  phase('Repair')
  const REPAIR = [
    COMMON,
    '',
    'Your label: fix-regress. You own the whole repository except DECISIONS.md, README.md and evals/**.',
    'Two reviewers examined the last batch of fixes (DECISIONS.md D-347 to D-366 plus the seams items in docs/wip/seams.md) and reported the findings below, each with a reproduction. Work test-first: for each finding, first write a regression test that fails for the reported reason. If you cannot make it fail, the finding is not reproduced: skip it and say so. Otherwise fix it with the smallest change that keeps the intent of the original decision, and keep the lucky baseline (python -m tianlong baseline --lucky --mock --strict: 19 marks) and tests/acceptance green.',
    'If a finding shows that one of the decisions D-347 to D-366 was itself wrong, say which and write the corrected decision sentence in your log.',
    '',
    'Findings (JSON):',
    JSON.stringify(todo, null, 1),

  repair = await agent(REPAIR, { label: 'fix-regress', phase: 'Repair' })
}

return { leftovers: { seams: left[0], evals: left[1] }, checked: reviews.map(r => r.checked), findings: all, repair }
